════════════════════════════════════════════════════════════════════

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 1: Vérifier Configuration (2 min)                       │
└────────────────────────────────────────────────────────────────┘
./script/check_production_ready.sh

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 2: Sécuriser Secrets (10 min)                           │
└────────────────────────────────────────────────────────────────┘
./script/secure_secrets.sh

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 3: Configurer PayDunya LIVE (5 min)                     │
└────────────────────────────────────────────────────────────────┘
# Dashboard PayDunya → Mode LIVE → Récupérer clés
echo "live_master_XXX" > ~/.secrets/paydunya-master-key-production.txt
echo "live_private_XXX" > ~/.secrets/paydunya-private-key-production.txt
echo "live_public_XXX" > ~/.secrets/paydunya-public-key-production.txt
echo "live_token_XXX" > ~/.secrets/paydunya-token-production.txt
chmod 600 ~/.secrets/paydunya-*-production.txt

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 4: Token Docker Hub (2 min)                             │
└────────────────────────────────────────────────────────────────┘
# https://hub.docker.com/settings/security
# → New Access Token → aa-production
echo "dckr_pat_VOTRE_TOKEN" > ~/.docker/aa-token.txt
chmod 600 ~/.docker/aa-token.txt

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 5: Configurer Secrets Production (5 min)                │
└────────────────────────────────────────────────────────────────┘
cp .kamal/secrets-production-TEMPLATE .kamal/secrets
nano .kamal/secrets
# Modifier:
# - PAYDUNYA_MODE=live
# - CACHE_ENABLED=true
# - Vérifier chemins fichiers

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 6: Tester (2 min)                                       │
└────────────────────────────────────────────────────────────────┘
kamal secrets fetch -c config/deploy.production.yml
# → Doit afficher toutes les variables sans erreur

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 7: DÉPLOYER ! (10 min)                                  │
└────────────────────────────────────────────────────────────────┘
kamal setup -c config/deploy.production.yml

┌────────────────────────────────────────────────────────────────┐
│ ÉTAPE 8: Vérifier (2 min)                                     │
└────────────────────────────────────────────────────────────────┘
curl https://aa.okemamy.com/up
kamal logs -f -c config/deploy.production.yml
